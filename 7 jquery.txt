4. Write jQuery to show the use of click(), hover(), on(), trigger(),off() events:

<!DOCTYPE html>
<html>
<head>
    <title>Basic jQuery</title>

    <script src="jquery-3.6.1.min.js"></script>

    <script>
        $(document).ready(function () {

            $("#b1").hover(function () {
                document.write("Hello Aayush");
            });

            $("p").on("click", function () {
                $(this).css("background", "yellow");
            });

            $("#b2").click(function () {
                $("p").off("click");
            });

            $("#b3").click(function () {
                $("#t1").hide();
            });

            $("input").select(function () {
                $("input").after("Text Marked!");
            });

            $("#b4").click(function () {
                $("input").trigger("select");
            });

        });
    </script>
</head>

<body>

    <button id="b1">Hover</button><br>

    <p>Welcome to Mulund College</p><br>

    <p>Aayush: 239720</p><br>

    <button id="b2">Off</button><br>

    <p id="t1">Hello Aayush</p><br>

    <button id="b3">On</button><br>

    <input type="text" value="Aayush 239720"><br>

    <button id="b4">Trigger</button><br>

</body>
</html>
