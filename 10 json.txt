1. Write a program to create JSON.

<html>
<head>
<title>Creating JSON</title>
</head>

<body>

<script type="text/javascript">

     var data = {
         "name":"Romal",
         "age":20,
         "dept":"IT"
     }

     document.writeln(JSON.stringify(data) + "</br>");

     document.writeln(
         JSON.stringify(data,["name","age","dept"]) + "<br/>"
     );

     document.writeln(
         JSON.stringify(data,["name","age","dept"],5)
     );

</script>

</body>
</html>
