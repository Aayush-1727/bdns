Write a jQuery to get and set text context of the elements.:

<!DOCTYPE html>
<head>
    <title>Basic jQuery</title>

    <script src="jquery-3.6.1.min.js"></script>

    <script>
      $(document).ready(function(){

         $("#b1").click(function(){
            var str = $("p").text();
            alert(str);
         })

         $("#b2").click(function(){
            $("p").text("Hello World");
         })

      })
   </script>
</head>

<body>
     <p>Romal Shah:239720</p>
     <p> MongoDB</p>

     <button id="b1">Get</button>

<!DOCTYPE html>
<head>
    <title>Basic jQuery</title>

    <script src="jquery-3.6.1.min.js"></script>

    <script>
      $(document).ready(function(){

         $("#b1").click(function(){
            var str = $("p").html();

         })

   </script>
